+export async function bulkUpdateVerificationStatus(
+  _parent: unknown,
+  args: { businessIds: string[], status: string },
+  context: { headers: { authorization?: string } }
+): Promise<{
+  success: boolean;
+  updatedCount: number;
+  error?: string;
+}> {
+  const { businessIds, status } = args;
+  const authHeader = context.headers.authorization;
+
+  // Validate input
+  if (!businessIds || businessIds.length === 0) {
+    return {
+      success: false,
+      updatedCount: 0,
+      error: "Business IDs required",
+    };
+  }
+
+  // Validate status
+  const validStatuses = ['pending', 'approved', 'flagged'];
+  if (!validStatuses.includes(status)) {
+    return {
+      success: false,
+      updatedCount: 0,
+      error: "Invalid status. Must be 'pending', 'approved', or 'flagged'",
+    };
+  }
+
+  // Extract JWT token from Authorization header
+  if (!authHeader || !authHeader.startsWith("Bearer ")) {
+    return {
+      success: false,
+      updatedCount: 0,
+      error: "Authorization required",
+    };
+  }
+
+  const token = authHeader.substring(7);
+
+  // Verify token and extract user ID
+  let payload: JwtPayload;
+  try {
+    payload = verifyToken(token);
+  } catch (error) {
+    return {
+      success: false,
+      updatedCount: 0,
+      error: "Invalid or expired token",
+    };
+  }
+
+  // Verify user is admin
+  if (payload.role !== 'admin') {
+    return {
+      success: false,
+      updatedCount: 0,
+      error: "Admin access required",
+    };
+  }
+
+  // Get database client and perform bulk update
+  const client = getPool();
+  try {
+    const updatedBusinesses = await bulkUpdateVerificationStatus(client, businessIds, status as 'pending' | 'approved' | 'flagged');
+    return {
+      success: true,
+      updatedCount: updatedBusinesses.length,
+    };
+  } catch (error) {
+    return {
+      success: false,
+      updatedCount: 0,
+      error: error instanceof Error ? error.message : "Update failed",
+    };
+  }

+/**
+ * Finds all potential duplicate businesses from a list of existing businesses.
+ * Compares the searched business against each existing business and returns
+ * all matches with their similarity scores, identifying the best match.
+ *
+ * @param searchedBusiness - The business to search for duplicates
+ * @param existingBusinesses - List of existing businesses to compare against
+ * @param config - Configuration thresholds (uses defaults if not provided)
+ * @returns DuplicateSearchResult with all matches and best match identified
+ */
+export function findDuplicates(
+  searchedBusiness: BusinessForComparison,
+  existingBusinesses: BusinessForComparison[],
+  config: DuplicateDetectionConfig = DEFAULT_DUPLICATE_CONFIG
+): DuplicateSearchResult {
+  const matches: DuplicateMatch[] = [];
+
+  for (const existing of existingBusinesses) {
+    const checkResult = checkForDuplicate(searchedBusiness, existing, config);
+
+    // Only include matches that meet the duplicate criteria
+    if (checkResult.isPotentialDuplicate) {
+      matches.push({
+        business: existing,
+        isPotentialDuplicate: true,
+        nameSimilarity: checkResult.nameSimilarity,
+        addressSimilarity: checkResult.addressSimilarity,
+        combinedScore: checkResult.combinedScore,
+      });
+    }
+  }
+
+  // Sort matches by combined score (highest first)
+  matches.sort((a, b) => b.combinedScore - a.combinedScore);
+
+  // The best match is the one with the highest combined score
+  const bestMatch = matches.length > 0 ? matches[0] : null;
+
+  return {
+    searchedBusiness,
+    matches,
+    bestMatch,
+    matchCount: matches.length,
+  };
